Kompletný systém: Model, AI a Klient-side efekty

local workspace = game:GetService("Workspace")
local starterPlayer = game:GetService("StarterPlayer")
local lighting = game:GetService("Lighting")

-- Vymaže starého hada a staré efekty
if workspace:FindFirstChild("ModernSnakeNPC") then workspace.ModernSnakeNPC:Destroy() end
if starterPlayer.StarterPlayerScripts:FindFirstChild("SnakeHorrorEffects") then starterPlayer.StarterPlayerScripts.SnakeHorrorEffects:Destroy() end
if lighting:FindFirstChild("SnakeRedGlow") then lighting.SnakeRedGlow:Destroy() end

local snakeModel = Instance.new("Model")
snakeModel.Name = "ModernSnakeNPC"
snakeModel.Parent = workspace

-- Vytvorenie Humanoida (rýchlosť necháme na 15, dá sa utiecť)
local humanoid = Instance.new("Humanoid")
humanoid.RequiresNeck = false
humanoid.HipHeight = 2
humanoid.MaxHealth = 500
humanoid.Health = 500
humanoid.WalkSpeed = 15
humanoid.Parent = snakeModel

local numSegments = 20
local headSize = 4.5
local bodySize = 2.2
local overlap = 0.85

-- Vytvorenie HumanoidRootPart
local hrp = Instance.new("Part")
hrp.Name = "HumanoidRootPart"
hrp.Size = Vector3.new(3, 3, 3)
hrp.Transparency = 1
hrp.CanCollide = false
hrp.Position = Vector3.new(0, 10, 0)
hrp.Parent = snakeModel
snakeModel.PrimaryPart = hrp

-- Vytvorenie Hlavy (Smrtiaca časť)
local head = Instance.new("Part")
head.Name = "Head"
head.Shape = Enum.PartType.Ball
head.Size = Vector3.new(headSize, headSize, headSize)
head.Color = Color3.fromRGB(255, 255, 255)
head.Material = Enum.Material.SmoothPlastic
head.CFrame = hrp.CFrame
head.CustomPhysicalProperties = PhysicalProperties.new(0.5, 0.3, 0.5)
head.Parent = snakeModel

local face = Instance.new("Decal")
face.Texture = "rbxassetid://8320404396"
face.Face = Enum.NormalId.Front
face.Parent = head

local weld = Instance.new("WeldConstraint")
weld.Part0 = hrp
weld.Part1 = head
weld.Parent = hrp

local parts = {head}

-- Vytvorenie Tela
for i = 1, numSegments do
    local segment = Instance.new("Part")
    segment.Name = "Body_" .. i
    segment.Shape = Enum.PartType.Ball
    segment.Size = Vector3.new(bodySize, bodySize, bodySize)
    segment.Color = Color3.fromRGB(10, 10, 10)
    segment.Material = Enum.Material.SmoothPlastic
    segment.Massless = true
    
    local prevPart = parts[i]
    local offsetDist = (prevPart.Size.Z / 2) + (segment.Size.Z / 2)
    segment.CFrame = prevPart.CFrame * CFrame.new(0, 0, offsetDist * overlap)
    segment.Parent = snakeModel
    
    table.insert(parts, segment)
end

-- Pospojovanie guličiek
for i = 1, #parts - 1 do
    local part0 = parts[i]
    local part1 = parts[i+1]
    
    local offset0 = (part0.Size.Z / 2) * overlap
    local offset1 = (part1.Size.Z / 2) * overlap
    
    local att0 = Instance.new("Attachment")
    att0.Position = Vector3.new(0, 0, offset0)
    att0.Parent = part0
    
    local att1 = Instance.new("Attachment")
    att1.Position = Vector3.new(0, 0, -offset1)
    att1.Parent = part1
    
    local ballSocket = Instance.new("BallSocketConstraint")
    ballSocket.Attachment0 = att0
    ballSocket.Attachment1 = att1
    ballSocket.LimitsEnabled = true
    ballSocket.UpperAngle = 50
    ballSocket.TwistLimitsEnabled = true
    ballSocket.Parent = part0
    
    local noCollision = Instance.new("NoCollisionConstraint")
    noCollision.Part0 = part0
    noCollision.Part1 = part1
    noCollision.Parent = part0
end

-- AI SKRIPT NA PRENASLEDOVANIE A ZABÍJANIE
local aiScript = Instance.new("Script")
aiScript.Name = "KillerAI"
aiScript.Source = [[
local npc = script.Parent
local humanoid = npc:WaitForChild("Humanoid")
local hrp = npc:WaitForChild("HumanoidRootPart")
local head = npc:WaitForChild("Head")

local function getNearestPlayer()
    local nearestTarget = nil
    local shortestDistance = math.huge
    
    for _, player in pairs(game.Players:GetPlayers()) do
        if player.Character and player.Character:FindFirstChild("HumanoidRootPart") and player.Character:FindFirstChild("Humanoid").Health > 0 then
            local distance = (player.Character.HumanoidRootPart.Position - hrp.Position).Magnitude
            if distance < shortestDistance then
                shortestDistance = distance
                nearestTarget = player.Character.HumanoidRootPart
            end
        end
    end
    return nearestTarget
end

task.spawn(function()
    while task.wait(0.1) do
        local target = getNearestPlayer()
        if target then
            humanoid:MoveTo(target.Position)
        end
    end
end)

head.Touched:Connect(function(hit)
    local character = hit.Parent
    if character and character:FindFirstChild("Humanoid") and character.Name ~= npc.Name then
        character.Humanoid.Health = 0 -- ZABITIE
    end
end)
]]
aiScript.Parent = snakeModel

-- KLIENTSKÝ SKRIPT (HOROROVÉ EFEKTY)
local effectsScript = Instance.new("LocalScript")
effectsScript.Name = "SnakeHorrorEffects"
effectsScript.Source = [[
local runService = game:GetService("RunService")
local players = game:GetService("Players")
local workspace = game:GetService("Workspace")
local lighting = game:GetService("Lighting")

local player = players.LocalPlayer
local camera = workspace.CurrentCamera

local gui = Instance.new("ScreenGui")
gui.IgnoreGuiInset = true
gui.Parent = player:WaitForChild("PlayerGui")

local redFlash = Instance.new("Frame")
redFlash.Size = UDim2.new(1, 0, 1, 0)
redFlash.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
redFlash.BackgroundTransparency = 1
redFlash.ZIndex = 10
redFlash.Parent = gui

local colorCorrection = Instance.new("ColorCorrectionEffect")
colorCorrection.Name = "SnakeRedGlow"
colorCorrection.Parent = lighting

local heartbeatSound = Instance.new("Sound")
heartbeatSound.SoundId = "rbxassetid://138084661"
heartbeatSound.Looped = true
heartbeatSound.Volume = 0
heartbeatSound.Parent = workspace

local MAX_DISTANCE = 70

runService.RenderStepped:Connect(function()
    local char = player.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") or char.Humanoid.Health <= 0 then
        redFlash.BackgroundTransparency = 1
        colorCorrection.TintColor = Color3.fromRGB(255, 255, 255)
        colorCorrection.Contrast = 0
        heartbeatSound.Volume = 0
        return
    end

    local hrp = char.HumanoidRootPart
    local snake = workspace:FindFirstChild("ModernSnakeNPC")
    
    if snake and snake:FindFirstChild("HumanoidRootPart") then
        local distance = (snake.HumanoidRootPart.Position - hrp.Position).Magnitude
        if distance < MAX_DISTANCE then
            if not heartbeatSound.IsPlaying then heartbeatSound:Play() end
            local intensity = 1 - (distance / MAX_DISTANCE)
            local pulseSpeed = 5 + (intensity * 15)
            local pulse = (math.sin(tick() * pulseSpeed) + 1) / 2
            heartbeatSound.Volume = intensity * 3
            local flashStrength = intensity * 0.5 + (pulse * intensity * 0.3)
            redFlash.BackgroundTransparency = 1 - flashStrength
            local colorGlow = 255 - (intensity * 180)
            colorCorrection.TintColor = Color3.fromRGB(255, colorGlow, colorGlow)
            colorCorrection.Contrast = intensity * 0.4
            local shakeAmount = intensity * 0.6
            camera.CFrame = camera.CFrame * CFrame.Angles((math.random()-0.5)*shakeAmount, (math.random()-0.5)*shakeAmount, (math.random()-0.5)*shakeAmount)
        else
            redFlash.BackgroundTransparency = 1
            colorCorrection.TintColor = Color3.fromRGB(255, 255, 255)
            colorCorrection.Contrast = 0
            heartbeatSound.Volume = 0
        end
    else
        redFlash.BackgroundTransparency = 1
        colorCorrection.TintColor = Color3.fromRGB(255, 255, 255)
        colorCorrection.Contrast = 0
        heartbeatSound.Volume = 0
    end
end)
]]
effectsScript.Parent = starterPlayer.StarterPlayerScripts

print("Červené hororové blikanie pripravené! Stlač Play.")
